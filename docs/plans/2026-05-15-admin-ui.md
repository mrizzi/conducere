# Admin UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a web-based admin dashboard to Joinora with GitHub OAuth, role-based access (admin/viewer), session monitoring, and settings — all embedded in the existing FastAPI process.

**Architecture:** New `admin_web.py` module adds `/admin/` routes to the existing FastAPI app via `app.mount()`. GitHub OAuth flow with JWT session cookies. Role config via `admin_roles.json`. Frontend is a vanilla JS SPA in `joinora/admin_frontend/`. No participant removal — sessions are open by design.

**Tech Stack:** FastAPI, PyJWT, httpx (GitHub OAuth), vanilla HTML/CSS/JS

```python
import json
import os
import secrets
from datetime import datetime, timezone, timedelta
from pathlib import Path

import jwt
from fastapi import FastAPI, HTTPException, Request, Response
from fastapi.responses import FileResponse, RedirectResponse
from starlette.middleware.base import BaseHTTPMiddleware

from joinora.session_store import SessionStore

_JWT_ALGORITHM = "HS256"
_COOKIE_NAME = "joinora_admin"
_JWT_EXPIRY_HOURS = 24


def load_roles(path: Path) -> dict[str, list[str]]:
    if not path.exists():
        return {"admin": [], "viewer": []}
    with open(path) as f:
        data = json.load(f)
    return {
        "admin": data.get("admin", []),
        "viewer": data.get("viewer", []),
    }


def _resolve_role(roles: dict[str, list[str]], username: str) -> str | None:
    for role, users in roles.items():
        if username in users:
            return role
    return None
```

```python
from fastapi.staticfiles import StaticFiles


def create_admin_app(
    store: SessionStore,
    roles_path: Path,
    jwt_secret: str,
    github_client_id: str,
    github_client_secret: str,
    base_url: str,
) -> FastAPI:
    app = FastAPI()
    roles = load_roles(roles_path)

    _PUBLIC_PATHS = {"/admin/login", "/admin/callback"}

    class AdminAuthMiddleware(BaseHTTPMiddleware):
        async def dispatch(self, request: Request, call_next):
            if request.url.path in _PUBLIC_PATHS:
                return await call_next(request)
            if request.url.path.startswith("/admin/static"):
                return await call_next(request)

            token = request.cookies.get(_COOKIE_NAME)
            if not token:
                return RedirectResponse(url="/admin/login", status_code=307)
            try:
                payload = jwt.decode(token, jwt_secret, algorithms=[_JWT_ALGORITHM])
                request.state.username = payload["sub"]
                request.state.role = payload["role"]
            except (jwt.ExpiredSignatureError, jwt.InvalidTokenError):
                return RedirectResponse(url="/admin/login", status_code=307)
            return await call_next(request)

    app.add_middleware(AdminAuthMiddleware)

    @app.get("/admin/login")
    async def login_page():
        github_url = (
            f"https://github.com/login/oauth/authorize"
            f"?client_id={github_client_id}"
            f"&redirect_uri={base_url}/admin/callback"
            f"&scope=read:user"
        )
        return Response(
            content=(
                '<!DOCTYPE html><html><head><title>Joinora Admin Login</title>'
                "<style>"
                "body{background:#1a1a2e;color:#e0e0e0;font-family:system-ui;"
                "display:flex;align-items:center;justify-content:center;"
                "min-height:100vh;margin:0}"
                ".card{background:#2a2a4a;padding:48px;border-radius:12px;text-align:center}"
                "h1{margin:0 0 8px;font-size:24px}"
                "p{color:#94a3b8;margin:0 0 32px}"
                "a{display:inline-block;background:#238636;color:#fff;padding:12px 24px;"
                "border-radius:6px;text-decoration:none;font-weight:500}"
                "a:hover{background:#2ea043}"
                "</style></head>"
                '<body><div class="card"><h1>Joinora Admin</h1>'
                "<p>Sign in to continue</p>"
                f'<a href="{github_url}">Sign in with GitHub</a>'
                "</div></body></html>"
            ),
            media_type="text/html",
        )

    @app.get("/admin/callback")
    async def oauth_callback(code: str):
        import httpx

        async with httpx.AsyncClient() as client:
            token_resp = await client.post(
                "https://github.com/login/oauth/access_token",
                json={
                    "client_id": github_client_id,
                    "client_secret": github_client_secret,
                    "code": code,
                },
                headers={"Accept": "application/json"},
            )
            access_token = token_resp.json().get("access_token")
            if not access_token:
                raise HTTPException(status_code=401, detail="OAuth failed")

            user_resp = await client.get(
                "https://api.github.com/user",
                headers={"Authorization": f"Bearer {access_token}"},
            )
            username = user_resp.json().get("login")

        role = _resolve_role(roles, username)
        if role is None:
            raise HTTPException(status_code=403, detail="Not authorized")

        exp = datetime.now(timezone.utc) + timedelta(hours=_JWT_EXPIRY_HOURS)
        jwt_token = jwt.encode(
            {"sub": username, "role": role, "exp": exp},
            jwt_secret,
            algorithm=_JWT_ALGORITHM,
        )
        response = RedirectResponse(url="/admin/", status_code=307)
        response.set_cookie(
            _COOKIE_NAME,
            jwt_token,
            httponly=True,
            samesite="lax",
            max_age=_JWT_EXPIRY_HOURS * 3600,
        )
        return response

    @app.post("/admin/logout")
    async def logout():
        response = RedirectResponse(url="/admin/login", status_code=307)
        response.delete_cookie(_COOKIE_NAME)
        return response

    def _require_admin(request: Request) -> None:
        if request.state.role != "admin":
            raise HTTPException(status_code=403, detail="Admin role required")

    @app.get("/admin/api/stats")
    async def get_stats(request: Request):
        sessions = store.list_all_sessions()
        active = [s for s in sessions if s.status.value == "active"]
        complete = [s for s in sessions if s.status.value == "complete"]
        total_messages = sum(len(s.messages) for s in sessions)
        all_participants = set()
        for s in sessions:
            for p in s.participants:
                all_participants.add(p.name)
        return {
            "active_sessions": len(active),
            "completed_sessions": len(complete),
            "total_messages": total_messages,
            "total_participants": len(all_participants),
        }

    admin_frontend_dir = Path(__file__).parent / "admin_frontend"

    @app.get("/admin/")
    async def admin_spa(request: Request):
        index = admin_frontend_dir / "index.html"
        if index.exists():
            return FileResponse(str(index))
        raise HTTPException(status_code=404, detail="Admin frontend not found")

    if admin_frontend_dir.exists():
        app.mount(
            "/admin/static",
            StaticFiles(directory=str(admin_frontend_dir)),
            name="admin-static",
        )

    return app
```

```python
    @app.get("/admin/api/sessions")
    async def list_sessions(
        request: Request,
        status: str | None = None,
        q: str | None = None,
    ):
        sessions = store.list_all_sessions()
        if status:
            sessions = [s for s in sessions if s.status.value == status]
        if q:
            q_lower = q.lower()
            sessions = [s for s in sessions if q_lower in s.title.lower()]
        return {
            "sessions": [
                {
                    "id": s.id,
                    "title": s.title,
                    "status": s.status.value,
                    "participant_count": len(s.participants),
                    "message_count": len(s.messages),
                    "created_at": s.created_at.isoformat(),
                }
                for s in sessions
            ]
        }

    @app.get("/admin/api/sessions/{session_id}")
    async def get_session_detail(request: Request, session_id: str):
        session = store.get_session(session_id)
        if session is None:
            raise HTTPException(status_code=404, detail="Session not found")
        return {
            "id": session.id,
            "title": session.title,
            "status": session.status.value,
            "created_at": session.created_at.isoformat(),
            "participants": [
                {
                    "name": p.name,
                    "last_seen": p.last_seen.isoformat() if p.last_seen else None,
                }
                for p in session.participants
            ],
            "messages": [m.model_dump(mode="json") for m in session.messages],
        }

    @app.post("/admin/api/sessions/{session_id}/end")
    async def end_session(request: Request, session_id: str):
        _require_admin(request)
        try:
            result = store.end_session(session_id)
        except ValueError as e:
            raise HTTPException(status_code=404, detail=str(e))
        return result

    @app.post("/admin/api/sessions/{session_id}/reopen")
    async def reopen_session(request: Request, session_id: str):
        _require_admin(request)
        try:
            store.reopen_session(session_id)
        except ValueError as e:
            raise HTTPException(status_code=400, detail=str(e))
        return {"status": "active"}
```

```python
    async def update_roles(request: Request):
        _require_admin(request)
        body = await request.json()
        new_roles = {
            "admin": body.get("admin", []),
            "viewer": body.get("viewer", []),
        }
        with open(roles_path, "w") as f:
            json.dump(new_roles, f, indent=2)
        nonlocal roles
        roles = new_roles
        return new_roles

    @app.get("/admin/api/oauth-status")
    async def get_oauth_status(request: Request):
        _require_admin(request)
        masked_id = github_client_id[:4] + "***" if github_client_id else None
        return {
            "configured": bool(github_client_id and github_client_secret),
            "client_id": masked_id,
            "callback_url": f"{base_url}/admin/callback",
        }
```
